cmake_minimum_required(VERSION 3.25)
project(tessitura VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 26)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
find_package(SQLite3 REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(Boost CONFIG REQUIRED)
add_executable(tessitura-server src/main.cpp)
target_include_directories(tessitura-server PRIVATE include ${Boost_INCLUDE_DIRS})
target_link_libraries(tessitura-server PRIVATE SQLite::SQLite3 OpenSSL::Crypto Threads::Threads)
target_compile_options(tessitura-server PRIVATE -Wall -Wextra -Wpedantic)
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME tessitura.http.e2e COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_api.py $<TARGET_FILE:tessitura-server> ${CMAKE_SOURCE_DIR})
  set_tests_properties(tessitura.http.e2e PROPERTIES TIMEOUT 35)
endif()
